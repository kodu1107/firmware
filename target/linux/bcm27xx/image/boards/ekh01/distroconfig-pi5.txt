################################################################################
# OpenMANET Raspberry Pi 5 bootloader configuration
################################################################################

[all]
disable_overscan=1

# Keep the primary UART available for recovery and hardware debugging.
enable_uart=1

# Keep the legacy framebuffer stable with HDMI capture devices. The Pi 5 KMS
# overlay can drop the signal during driver handoff with some capture cards.
hdmi_force_hotplug=1
hdmi_group=1
hdmi_mode=16
hdmi_drive=2

# The Pi 5 product overlays are written directly to config.txt by
# Build/boot-ekh01-pi5. Do not enable the legacy BCM283x SPI controller here;
# the MM6108 overlay enables the RP1 SPI controller used by the 40-pin header.
camera_auto_detect=1
